Reverse names and inverse calls

classical class Names {
  state long count = 2;

  rev void step() { count += 1; }

  long value(long reverse) {
    reverse -= 1;
    return reverse;
  }

  entry void main() {
    long reverse = 2;
    reverse = value(reverse);
    assert(reverse == 1);
    reverse step();
    reverse { step(); }
    assert(count == 0);
  }
}

classical class Types {
  variant reverse { case Value(long value); }

  entry void main() {
    reverse held = new reverse.Value(3);
    match (held) {
      case reverse.Value(long value) { assert(value == 3); }
    }
  }
}
